def calculate_position_size(signal_confidence: float, account_balance: float, 
                          market_volatility: float, max_risk_pct: float = 0.1) -> float:
    """
    Calculate optimal position size for real money trading
    
    Args:
        signal_confidence: AI confidence score (0.0-1.0)
        account_balance: Current USDT balance from Binance
        market_volatility: Recent BTC volatility measure
        max_risk_pct: Maximum risk per trade (default 10%)
    
    Returns:
        Position size in USDT
    """
    # Base position size calculation
    base_size = account_balance * max_risk_pct
    
    # Confidence adjustment (higher confidence = larger position)
    confidence_multiplier = min(signal_confidence * 1.5, 1.0)
    
    # Volatility adjustment (higher volatility = smaller position)
    volatility_multiplier = max(1.0 - market_volatility, 0.3)
    
    # Final position size
    position_size = base_size * confidence_multiplier * volatility_multiplier
    
    # Minimum and maximum bounds
    return max(min(position_size, account_balance * 0.15), 50.0)  # Min $50, Max 15%
```

class RealTimeRiskMonitor:
    """Monitor AI-driven trading positions for risk management"""
    
    def __init__(self, max_drawdown: float = 0.05, max_daily_loss: float = 0.10):
        self.max_drawdown = max_drawdown  # 5% max drawdown
        self.max_daily_loss = max_daily_loss  # 10% max daily loss
        
    async def evaluate_risk(self, current_positions: List[Position], 
                          account_balance: float) -> RiskAction:
        """Evaluate current risk and return recommended action"""
        
        # Calculate current drawdown
        total_unrealized_pnl = sum(pos.unrealized_pnl for pos in current_positions)
        current_drawdown = abs(total_unrealized_pnl) / account_balance
        
        # Risk-based AI decision
        if current_drawdown > self.max_drawdown:
            return RiskAction.CLOSE_ALL_POSITIONS
        elif current_drawdown > self.max_drawdown * 0.7:
            return RiskAction.REDUCE_POSITION_SIZES
        else:
            return RiskAction.CONTINUE_NORMAL
```

class BTCTradingLoop:
    """Main trading loop for BTC-only AI trading"""
    
    def __init__(self):
        self.binance = BinanceBTCSpotClient()
        self.position_manager = BTCPositionManager()
        self.alerts = AlertsManager()
        
    async def process_ai_signal(self, signal_data: dict):
        """Process AI signal for BTC trading"""
        
        # 1. Validate signal
        if signal_data['symbol'] != 'BTCUSDT':
            return {'status': 'rejected', 'reason': 'Only BTCUSDT supported'}
            
        # 2. Check safety limits
        if not self.check_trading_limits():
            return {'status': 'rejected', 'reason': 'Trading limits exceeded'}
            
        # 3. Execute trade based on signal
        if signal_data['type'] == 'BUY':
            return await self.execute_btc_buy(signal_data)
        elif signal_data['type'] == 'SELL':
            return await self.execute_btc_sell(signal_data)
            
    async def execute_btc_buy(self, signal_data: dict):
        """Execute BTC buy order"""
        
        # Calculate position size
        usdt_balance = self.binance.get_usdt_balance()
        position_size = min(
            float(os.getenv('MAX_POSITION_SIZE_USDT', '100')),
            usdt_balance * 0.95  # Use 95% max to account for fees
        )
        
        # Place market buy order
        order_result = self.binance.buy_btc_market(position_size)
        
        # Track position
        self.position_manager.open_position(order_result, signal_data)
        
        # Send alert
        await self.alerts.send_trade_alert('BUY', order_result)
        
        return order_result

class BTCPositionManager:
    """Manage BTC spot positions with TP/SL tracking"""
    
    def __init__(self):
        self.current_position = None
        self.position_history = []
        
    def open_position(self, order_result: dict, signal_data: dict):
        """Track new BTC position"""
        
        btc_quantity = float(order_result['executedQty'])
        avg_price = float(order_result['fills'][0]['price'])
        
        self.current_position = {
            'symbol': 'BTCUSDT',
            'side': 'LONG',  # We only buy BTC with USDT
            'quantity': btc_quantity,
            'entry_price': avg_price,
            'entry_time': datetime.now(),
            'stop_loss': avg_price * (1 - BTC_TRADING_CONFIG['STOP_LOSS_PCT'] / 100),
            'take_profit': avg_price * (1 + BTC_TRADING_CONFIG['TAKE_PROFIT_PCT'] / 100),
            'ai_confidence': signal_data.get('confidence', 0.0),
            'order_id': order_result['orderId']
        }
        
        # Save to DynamoDB
        self.save_position_to_db()
        
    def check_tp_sl(self, current_btc_price: float) -> dict:
        """Check if position should be closed due to TP/SL"""
        
        if not self.current_position:
            return {'action': 'none'}
            
        # Check Stop Loss
        if current_btc_price <= self.current_position['stop_loss']:
            return {
                'action': 'close',
                'reason': 'stop_loss',
                'target_price': self.current_position['stop_loss']
            }
            
        # Check Take Profit
        if current_btc_price >= self.current_position['take_profit']:
            return {
                'action': 'close',
                'reason': 'take_profit',
                'target_price': self.current_position['take_profit']
            }
            
        return {'action': 'hold'}
        
    def close_position(self, sell_order_result: dict, reason: str):
        """Close current BTC position and calculate P&L"""
        
        if not self.current_position:
            return
            
        exit_price = float(sell_order_result['fills'][0]['price'])
        exit_time = datetime.now()
        
        # Calculate P&L
        entry_value = self.current_position['quantity'] * self.current_position['entry_price']
        exit_value = self.current_position['quantity'] * exit_price
        pnl_usdt = exit_value - entry_value
        pnl_pct = (pnl_usdt / entry_value) * 100
        
        # Store in history
        closed_position = {
            **self.current_position,
            'exit_price': exit_price,
            'exit_time': exit_time,
            'pnl_usdt': pnl_usdt,
            'pnl_percentage': pnl_pct,
            'close_reason': reason,
            'hold_duration': (exit_time - self.current_position['entry_time']).total_seconds()
        }
        
        self.position_history.append(closed_position)
        self.current_position = None
        
        # Save to DynamoDB
        self.save_closed_position_to_db(closed_position)
        
        return closed_position
